📦 GitSandboxCoordinator & WorkspaceDriver Class Model 
interface WorkspaceDriver {
  initialize(runId: string, baseRef: string): Promise<WorkspaceContext>;
  writeFile(relPath: string, content: string): Promise<void>;
  readFile(relPath: string): Promise<string>;
  commitStep(step: number, fitness: number): Promise<string>;
  rollbackStep(): Promise<void>; // <30ms git reset --hard
  promote(targetBranch: string): Promise<void>;
  teardown(): Promise<void>;
  getWorkingDir(): string;
}

class GitSandboxCoordinator {
  private driver: WorkspaceDriver;
  private lockfilePath: string = ".perfgym/sandbox.lock";

  async acquire(runId: string): Promise<WorkspaceContext> {
    await this.acquireLockfile(runId);
    return this.driver.initialize(runId, "HEAD");
  }

  async rejectStep(): Promise<void> {
    // Discards uncommitted agent edits immediately
    await this.driver.rollbackStep();
  }

  async acceptStep(step: number, score: number): Promise<StepReceipt> {
    const sha = await this.driver.commitStep(step, score);
    return { step, commitSha: sha, status: "COMMITTED" };
  }
}

🔍 AstLinterEngine with @ast-grep/napi Direct Rust Binding 
interface RuleDefinition {
  id: string;
  message: string;
  severity: "error" | "warning";
  penalty: number;
  action: "disqualify" | "penalize";
  pattern: string;
}

class AstLinterEngine {
  private rules: RuleDefinition[] = [];

  // Scans in-process via Tree-Sitter napi binding in <10ms
  async scan(files: string[], worktreeDir: string): Promise<LintResult> {
    let totalPenalty = 0;
    const matches: RuleMatch[] = [];

    for (const file of files) {
      const ast = parseTreeSitter(readFile(file));
      for (const rule of this.rules) {
        const hits = ast.findAll(rule.pattern);
        if (hits.length > 0) {
          if (rule.action === "disqualify") {
            return { passed: false, disqualified: true, totalPenalty: Infinity };
          }
          totalPenalty += rule.penalty * hits.length;
          matches.push(...formatHits(hits, rule, file));
        }
      }
    }
    return { passed: totalPenalty === 0, disqualified: false, totalPenalty, matches };
  }
}

perfgym run --iterations 15 --model "anthropic/claude-3-7-sonnet" drives autonomous optimization using a Negative Memory Session Cache: 

class AutonomousHillClimbingRunner {
  private negativeMemory: NegativeMemoryCache;

  async runStep(stepNumber: number) {
    // 1. Synthesize prompt with history of failed attempts
    const prompt = this.synthesizer.buildPrompt({
      target: this.target,
      baseline: this.baseline,
      negativeMemory: this.negativeMemory.getFailures()
    });

    // 2. Query LLM (via direct Vercel AI SDK or local agent CLI)
    const mutation = await this.llmDriver.complete(prompt);

    // 3. Evaluate in isolated worktree
    const result = await this.pipeline.evaluate(mutation);

    // 4. Update negative memory to prevent cyclic looping
    if (!result.accepted) {
      this.negativeMemory.recordFailure(mutation.summary, result.reason);
    }
  }
}
